name: MoonCat v1.0.3 Unsigned Release Builds

on:
  push:
    branches: [build/mooncat-v102-android-verify]
    paths: [.github/workflows/mooncat-unsigned-release.yml]
  workflow_dispatch:

permissions:
  contents: read

jobs:
  releases:
    runs-on: ubuntu-latest
    timeout-minutes: 60
    env:
      CI_UNSIGNED_RELEASE: "true"
      DOVI_NATIVE_ENABLED: "false"
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"
          cache: gradle
      - name: Set up Android SDK
        shell: bash
        run: |
          SDKMANAGER="$(command -v sdkmanager || true)"
          if [ -z "$SDKMANAGER" ]; then
            SDKMANAGER="$(find "${ANDROID_HOME}/cmdline-tools" -type f -name sdkmanager -print -quit)"
          fi
          test -x "$SDKMANAGER"
          "$SDKMANAGER" "platforms;android-36" "build-tools;36.0.0"

      - name: Run MoonCat source, News, Studios and audio tests
        shell: bash
        run: |
          chmod +x gradlew
          ./gradlew :app:testFullReleaseUnitTest \
            --tests "com.nuvio.tv.core.mooncat.*" \
            --tests "com.nuvio.tv.core.catalog.*" \
            --tests "com.nuvio.tv.ui.screens.player.AudioLanguageChoicesTest" \
            --tests "com.nuvio.tv.ui.screens.player.AudioSourceIdentityTest" \
            --no-daemon --max-workers=1 --console=plain -Dorg.gradle.jvmargs="-Xmx4g -Dfile.encoding=UTF-8"

      - name: Compile unsigned TV and Android releases
        run: |
          ./gradlew :app:assembleFullRelease :mobile:assembleRelease \
            -x lintVitalFullRelease -x lintVitalAnalyzeFullRelease -x lintVitalReportFullRelease \
            --no-daemon --max-workers=1 --console=plain -Dorg.gradle.jvmargs="-Xmx4g -Dfile.encoding=UTF-8"

      - name: Verify both release packages
        shell: bash
        run: |
          set -euo pipefail
          TOOL="$ANDROID_HOME/build-tools/36.0.0"
          TV="$(find app/build/outputs/apk/full/release -name '*unsigned*.apk' -print -quit)"
          MOBILE="$(find mobile/build/outputs/apk/release -name '*unsigned*.apk' -print -quit)"
          test -s "$TV"; test -s "$MOBILE"
          "$TOOL/aapt" dump badging "$TV" | grep -E "package: name='dev.mooncat.tv' versionCode='10004'"
          "$TOOL/aapt" dump badging "$MOBILE" | grep -E "package: name='dev.mooncat.mobile' versionCode='10004'"
          sha256sum "$TV" "$MOBILE"

      - uses: actions/upload-artifact@v4
        with:
          name: MoonCat-TV-v1.0.3-unsigned
          path: app/build/outputs/apk/full/release/*unsigned*.apk
          retention-days: 14

      - uses: actions/upload-artifact@v4
        with:
          name: MoonCat-Mobile-v1.0.3-unsigned
          path: mobile/build/outputs/apk/release/*unsigned*.apk
          retention-days: 14
